import os
import json

file_to_id = {}
for file in os.listdir():
        if not file.endswith(".java"):
                continue
        file_to_id[file] = ""

with open("soids.txt") as f:
        for line in f:
                if line.strip() == "":
                        break
                nameentrypair = line.split(sep=":",maxsplit=1)
                name = nameentrypair[0]
                entry = nameentrypair[1].replace(" ", "").strip()
                if entry.startswith("//"):
                        entry = entry[2:].strip()
                if entry[:3].lower() == "id=":
                        entry = entry[3:]
                if entry[:4].lower() == "url:":
                        entry = entry[4:]
                if entry[:len("http://stackoverflow.com/questions/")].lower() == "http://stackoverflow.com/questions/":
                        entry = entry[len("http://stackoverflow.com/questions/"):]
                        entry = entry.split(sep="/",maxsplit=1)[0]
                file_to_id[name] = entry

with open("soids.json", 'w') as json_file:
    json.dump(file_to_id, json_file, indent = 4, sort_keys=True)
